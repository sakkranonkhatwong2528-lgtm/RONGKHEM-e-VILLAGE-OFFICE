import supabaseClient from "./supabase-config.js"; import { requireAdmin, logout } from "./assets/js/auth.js"; // ======================================== // ตั้งค่าแต่ละเมนูให้ตรงกับตารางจริง // ======================================== const MODULES = { news: { table: "news", title: "title", detail: "content", image: "image_url" }, complaint: { table: "complaints", title: "title", detail: "detail", image: null }, elderly: { table: "elderly", title: "name", detail: "detail", image: null }, vulnerable: { table: "vulnerable", title: "name", detail: "detail", image: null }, project: { table: "projects", title: "title", detail: "description", image: "image_url" }, incident: { table: "incidents", title: "title", detail: "description", image: null }, activity: { table: "activities", title: "title", detail: "description", image: "image_url" } }; // ======================================== // ELEMENTS // ======================================== const menuSelect = document.querySelector("#menuSelect"); const crudForm = document.querySelector("#crudForm"); const recordId = document.querySelector("#recordId"); const titleInput = document.querySelector("#titleInput"); const detailInput = document.querySelector("#detailInput"); const imageInput = document.querySelector("#imageInput"); const currentImageUrl = document.querySelector("#currentImageUrl"); const imagePreview = document.querySelector("#imagePreview"); const imagePreviewContainer = document.querySelector("#imagePreviewContainer"); const tableBody = document.querySelector("#tableBody"); const resetBtn = document.querySelector("#resetBtn"); const logoutBtn = document.querySelector("#logoutBtn"); const adminEmail = document.querySelector("#adminEmail"); const formTitle = document.querySelector("#formTitle"); // ======================================== // MODULE ปัจจุบัน // ======================================== function getCurrentModule() { const moduleName = menuSelect.value; return MODULES[moduleName]; } // ======================================== // ตรวจสอบ LOGIN // ======================================== async function checkAdmin() { const user = await requireAdmin(); if (!user) return; adminEmail.textContent = user.email; } // ======================================== // โหลดข้อมูล // ======================================== async function loadData() { const module = getCurrentModule(); tableBody.innerHTML = ` กำลังโหลดข้อมูล... `; try { const { data, error } = await supabaseClient .from(module.table) .select("*") .order("id", { ascending:false }); if (error) throw error; renderTable( data || [], module ); } catch(error) { console.error(error); tableBody.innerHTML = ` โหลดข้อมูลไม่สำเร็จ `; } } // ======================================== // แสดงตาราง // ======================================== function renderTable( data, module ) { if (!data.length) { tableBody.innerHTML = ` ยังไม่มีข้อมูล `; return; } tableBody.innerHTML = data.map(item => { const title = item[module.title] || ""; const detail = item[module.detail] || ""; const image = module.image ? item[module.image] : null; return ` ${item.id} ${ image ? `` : "-" } ${escapeHtml(title)} ${escapeHtml(detail)} `; }).join(""); } // ======================================== // ป้องกัน HTML // ======================================== function escapeHtml(text) { return String(text) .replaceAll("&","&") .replaceAll("<","<") .replaceAll(">",">") .replaceAll('"',""") .replaceAll("'","'"); } // ======================================== // เพิ่ม / แก้ไขข้อมูล // ======================================== crudForm.addEventListener( "submit", async event => { event.preventDefault(); const module = getCurrentModule(); const title = titleInput.value.trim(); const detail = detailInput.value.trim(); const payload = { [module.title]: title, [module.detail]: detail }; try { // =============================== // อัปโหลดรูป // =============================== if ( module.image && imageInput.files[0] ) { const file = imageInput.files[0]; const fileName = `${Date.now()}-${file.name}`; const { error: uploadError } = await supabaseClient .storage .from("images") .upload( fileName, file ); if ( uploadError ) throw uploadError; const { data: urlData } = supabaseClient .storage .from("images") .getPublicUrl( fileName ); payload[module.image] = urlData.publicUrl; } // =============================== // ใช้รูปเดิม // =============================== else if ( module.image && currentImageUrl.value ) { payload[module.image] = currentImageUrl.value; } // =============================== // UPDATE // =============================== if ( recordId.value ) { const { error } = await supabaseClient .from(module.table) .update(payload) .eq( "id", recordId.value ); if ( error ) throw error; alert( "แก้ไขข้อมูลเรียบร้อย" ); } // =============================== // INSERT // =============================== else { const { error } = await supabaseClient .from(module.table) .insert( [payload] ); if ( error ) throw error; alert( "บันทึกข้อมูลเรียบร้อย" ); } resetForm(); loadData(); } catch(error) { console.error(error); alert( "เกิดข้อผิดพลาด: " + error.message ); } } ); // ======================================== // CLICK EDIT / DELETE // ======================================== tableBody.addEventListener( "click", async event => { const editButton = event.target.closest( ".btn-edit" ); const deleteButton = event.target.closest( ".btn-delete" ); const module = getCurrentModule(); // =============================== // EDIT // =============================== if ( editButton ) { const id = editButton.dataset.id; const { data, error } = await supabaseClient .from(module.table) .select("*") .eq( "id", id ) .single(); if ( error ) { alert( error.message ); return; } recordId.value = data.id; titleInput.value = data[module.title] || ""; detailInput.value = data[module.detail] || ""; if ( module.image && data[module.image] ) { currentImageUrl.value = data[module.image]; imagePreview.src = data[module.image]; imagePreviewContainer .style .display = "block"; } formTitle.textContent = "✏️ แก้ไขข้อมูล"; window.scrollTo({ top:0, behavior:"smooth" }); } // =============================== // DELETE // =============================== if ( deleteButton ) { const id = deleteButton.dataset.id; const confirmDelete = confirm( "ต้องการลบข้อมูลนี้ใช่หรือไม่?" ); if ( !confirmDelete ) return; const { error } = await supabaseClient .from(module.table) .delete() .eq( "id", id ); if ( error ) { alert( error.message ); return; } alert( "ลบข้อมูลเรียบร้อย" ); loadData(); } } ); // ======================================== // PREVIEW รูป // ======================================== imageInput.addEventListener( "change", event => { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = e => { imagePreview.src = e.target.result; imagePreviewContainer .style .display = "block"; }; reader.readAsDataURL( file ); } ); // ======================================== // RESET FORM // ======================================== function resetForm() { crudForm.reset(); recordId.value = ""; currentImageUrl.value = ""; imagePreviewContainer .style .display = "none"; formTitle.textContent = "➕ เพิ่มข้อมูลใหม่"; } // ======================================== // RESET BUTTON // ======================================== resetBtn.addEventListener( "click", resetForm ); // ======================================== // เปลี่ยนเมนู // ======================================== menuSelect.addEventListener( "change", () => { resetForm(); loadData(); } ); // ======================================== // LOGOUT // ======================================== logoutBtn.addEventListener( "click", async () => { if ( confirm( "ต้องการออกจากระบบใช่หรือไม่?" ) ) { await logout(); } } ); // ======================================== // START SYSTEM // ======================================== await checkAdmin(); await loadData();
